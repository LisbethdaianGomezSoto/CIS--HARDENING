# ============================================================
# FW1 - FortiGate VM, FortiOS 7.0.9
# Configuracion inicial (consola CLI) antes de la auditoria
# La contrasena del usuario admin se define en el primer inicio de sesion.
# ============================================================
config system global
 set hostname FW1
 set admintimeout 5
 set pre-login-banner enable
end
config system replacemsg admin "pre_admin-disclaimer-text"
 set buffer "Acceso restringido. Lisbeth Gomez 2025-0701. Actividad monitoreada."
end
config system interface
 edit port1
  set alias "hacia-R1"
  set mode static
  set ip 10.7.2.1 255.255.255.252
  set allowaccess ping ssh https
 next
 edit port2
  set alias "hacia-NAT"
  set mode dhcp
  set allowaccess ping
 next
end
config router static
 edit 1
  set dst 10.7.1.0 255.255.255.0
  set gateway 10.7.2.2
  set device port1
 next
end
config firewall address
 edit "LAN-10.7.1.0"
  set subnet 10.7.1.0 255.255.255.0
 next
end
config firewall policy
 edit 1
  set name "LAN-a-Internet"
  set srcintf port1
  set dstintf port2
  set srcaddr "LAN-10.7.1.0"
  set dstaddr all
  set action accept
  set schedule always
  set service DNS HTTP HTTPS PING
  set nat enable
  set logtraffic all
 next
end
